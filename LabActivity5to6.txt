Code for Logarithmic Time
namespace Logarithmic_time
{
    public partial class Form1 : Form
    {
        public Form1()
        {
            InitializeComponent();
        }

        private void button1_Click(object sender, EventArgs e)
        {
            int[] numbers = { 10, 20, 30, 40, 50, 60, 70 };

            int left = 0;
            int right = numbers.Length - 1;

            string result = "Not Found";

            while (left <= right)
            {
                int middle = (left + right) / 2;

                if (numbers[middle] == 50)
                {
                    result = "Found!";
                    break;
                }
                else if (numbers[middle] < 50)
                {
                    left = middle + 1;
                }
                else
                {
                    right = middle - 1;
                }
            }

            lblResult1.Text = result;
        }

        private void button2_Click(object sender, EventArgs e)
        {
            int n = 16;
            string result = "";

            while (n > 1)
            {
                n = n / 2;
                result += n;
            }

            lblResult2.Text = result;
        }

        private void textBox1_TextChanged(object sender, EventArgs e)
        {

        }

        private void button3_Click(object sender, EventArgs e)
        {
            int low = 1;
            int high = 100;
            string result = "";

            while (low <= high)
            {
                int middle = (low + high) / 2;

                result += middle;

                high = middle - 1;
            }

            lblResult3.Text = result;
        }

        }
    }

Code for Exponential Time
namespace Exponential_time
{
    public partial class Form1 : Form
    {
        public Form1()
        {
            InitializeComponent();
        }

        private void button1_Click(object sender, EventArgs e)
        {
            int n = 3;
            int combinations = 1;

            for (int i = 0; i < n; i++)
            {
                combinations = combinations * 2;
            }

            lblResult1.Text = "Number of combinations: "
                + combinations;
        }

        private void button2_Click(object sender, EventArgs e)
        {
            int tosses = 5;
            int possibilities = 1;

            for (int i = 0; i < tosses; i++)
            {
                possibilities *= 2;
            }

            lblResult2.Text = "Possible results: "
                + possibilities;
        }

        private void textBox1_TextChanged(object sender, EventArgs e)
        {

        }

        private void button3_Click(object sender, EventArgs e)
        {
            string result = "";

            void Choose(int n, string combination)
            {
                if (n == 0)
                {
                    result += combination + Environment.NewLine;
                    return;
                }

                Choose(n - 1, combination + "A");
                Choose(n - 1, combination + "B");
            }

            Choose(3, "");

            lblResult3.Text = result;
        }

        private void label2_Click(object sender, EventArgs e)
        {

        }
    }
}

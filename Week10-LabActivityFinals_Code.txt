using System;
using System.Collections.Generic;

class LibraryManager
{
    private List<string> books = new List<string>();
    public void AddBook(string book)
    {
        books.Add(book);
    }
    public void BorrowBook(string book)
    {
        if (books.Contains(book))
        {
            books.Remove(book);
            Console.WriteLine(book + " borrowed successfully.");
        }
        else
        {
            Console.WriteLine("Book not available.");
        }
    }
    public void ReturnBook(string book)
    {
        books.Add(book);
        Console.WriteLine(book + " returned successfully.");
    }
    public void DisplayBooks()
    {
        foreach (string book in books)
        {
            Console.WriteLine(book);
        }
    }
}
class Program
{
    static void Main()
    {
        LibraryManager library = new LibraryManager();

        library.AddBook("Harry Potter");
        library.AddBook("The Little Prince");
        library.AddBook("Atomic Habits");

        library.BorrowBook("Harry Potter");

        library.ReturnBook("Harry Potter");

        Console.WriteLine(" Available Books:");
        library.DisplayBooks();
    }
}

using System;
class Node
{
    public string Data;
    public Node Next;

    public Node(string data)
    {
        Data = data;
        Next = null;
    }
}

class PatientWaitingList
{
    private Node head;
    private Node tail;
    public void AddPatient(string patient)
    {
        Node newNode = new Node(patient);

        if (head == null)
        {
            head = newNode;
            tail = newNode;
        }
        else
        {
            tail.Next = newNode;
            tail = newNode;
        }
    }
    public void DisplayPatients()
    {
        Node current = head;

        while (current != null)
        {
            Console.Write(current.Data + " -> ");
            current = current.Next;
        }

        Console.WriteLine("null");
    }
    public void SearchPatient(string patient)
    {
        Node current = head;

        while (current != null)
        {
            if (current.Data == patient)
            {
                Console.WriteLine("Patient found!");
                return;
            }

            current = current.Next;
        }

        Console.WriteLine("Patient not found.");
    }
    public void RemoveFirstPatient()
    {
        if (head != null)
        {
            Console.WriteLine("Served: " + head.Data);

            head = head.Next;

            if (head == null)
            {
                tail = null;
            }
        }
        else
        {
            Console.WriteLine("No patients waiting.");
        }
    }
}

class Program
{
    static void Main()
    {
        PatientWaitingList patients = new PatientWaitingList();

        patients.AddPatient("Patient 101");
        patients.AddPatient("Patient 102");
        patients.AddPatient("Patient 103");

        Console.WriteLine("Waiting List:");
        patients.DisplayPatients();

        patients.SearchPatient("Patient 102");

        patients.RemoveFirstPatient();

        Console.WriteLine("\nUpdated Waiting List:");
        patients.DisplayPatients();
    }
}

class DeliveryRoute
{
    private Node head;

    public DeliveryRoute()
    {
        head = new Node("Warehouse");
    }
    public void AddLocation(string location)
    {
        Node newNode = new Node(location);

        Node current = head;

        while (current.Next != null)
        {
            current = current.Next;
        }

        current.Next = newNode;
    }
    public void RemoveLocation(string location)
    {
        Node current = head;

        while (current.Next != null)
        {
            if (current.Next.Data == location)
            {
                current.Next = current.Next.Next;
                Console.WriteLine("Delivery removed.");
                return;
            }

            current = current.Next;
        }

        Console.WriteLine("Location not found.");
    }
    public void DisplayRoute()
    {
        Node current = head;

        while (current != null)
        {
            Console.Write(current.Data + " -> ");
            current = current.Next;
        }

        Console.WriteLine("null");
    }
    public void SearchDestination(string location)
    {
        Node current = head;

        while (current != null)
        {
            if (current.Data == location)
            {
                Console.WriteLine("Destination found!");
                return;
            }

            current = current.Next;
        }

        Console.WriteLine("Destination not found.");
    }
}

        {
            string book = cmbBooks.Text;

            if (books.Contains(book))

                books.Remove(book);

                lstResults.Items.Add(book + " borrowed successfully.");
            }
            else
            {
                MessageBox.Show("Book not available!");
            }
        }

        private void btnReturn_Click(object sender, EventArgs e)
        {
            string book = cmbBooks.Text;
            if (borrowedBooks.Contains(book))
            {
                borrowedBooks.Remove(book);
                books.Add(book);
                lstResults.Items.Add(book + " returned successfully.");
            }
            else
            {
                MessageBox.Show("Book was not borrowed!");
            }
